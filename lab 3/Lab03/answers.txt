c. def bubble_sort(self):
        n = len(self.array)

        for i in range(n - 1):
            swapped = False

            for j in range(n - 1 - i):
                if self.array[j] > self.array[j + 1]:
                    self.array[j], self.array[j + 1] = \
                        self.array[j + 1], self.array[j]
                    swapped = True
            if not swapped:
                break
